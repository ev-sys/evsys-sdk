```python
def generate(
    self,
    *,
    prompt: str,
    max_tokens: int = 256,
    temperature: float = 0.0,
    stop: list[str] | None = None,
) -> str:
    try:
        out = self.template.format(prompt=prompt)
    except (KeyError, IndexError):
        out = self.template
    if stop:
        for s in stop:
            idx = out.find(s)
            if idx >= 0:
                out = out[:idx]
    return out
```
